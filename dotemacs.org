#+begin_src emacs-lisp
  (defun nth-weekday-of-month-p (date &optional n)
  "Check if DATE is the Nth weekday of current month."
  (or n (setq n 1))
  (let* ((month (calendar-extract-month date))
         (year (calendar-extract-year date))
         (day (calendar-extract-day date))
         (month-days-to-n (mapcar
                           (lambda (day)
                              `(,month ,day ,year))
                           (number-sequence 1 day)))
         (business-days (seq-filter
                         (lambda (date)
                           (and (memq (calendar-day-of-week date)
                                      '(1 2 3 4 5))
                                (not (calendar-check-holidays date))))
                         month-days-to-n)))
    (and (eq n (length business-days))
         (eq day (calendar-extract-day (car (last business-days)))))))
#+end_src

    (defun ec/org-skip-habits-with-future-deadlines ()
    "Skip trees that are habits and have deadlines in the future."
    (let* ((is-habit (org-is-habit-p))  ; Check if it's a habit
           (deadline (org-entry-get nil "DEADLINE"))  ; Get deadline if it exists
           (deadline-time (and deadline (org-time-string-to-time deadline)))
           (now (current-time)))
      (if (and is-habit
               deadline-time
               (time-less-p now deadline-time))  ; Skip if deadline is in the future
          (save-excursion (or (outline-next-heading) (point-max)))  ; Skip the heading
        nil)))  ; Otherwise, don't skip

  (defun ec/org-skip-timed-entries ()
    "Skip agenda entries that have a time in their timestamp."
    (let* ((scheduled (org-entry-get nil "SCHEDULED"))
           (deadline (org-entry-get nil "DEADLINE"))
           (timestamp (or scheduled deadline)))
      (if (and timestamp
               (string-match "[0-9]\\{2\\}:[0-9]\\{2\\}" timestamp))  ; Check if time is present
          (save-excursion (or (outline-next-heading) (point-max)))  ; Skip this entry
        nil)))  ; Don't skip if no time is found

  (defun ec/org-skip-for-mission-control-agenda-habits ()
    "Skip function for mission control."
    (or (org-agenda-skip-entry-if 'todo
                             '("STICKY" "NEXT" "WIP" "SOON"))
        (and (not (org-is-habit-p))
             (save-excursion (or (outline-next-heading) (point-max))))
        (ec/org-skip-timed-entries)
        (ec/org-skip-habits-with-future-deadlines)))

  (defun ec/org-skip-for-mission-control-agenda-scheduled ()
    "Skip function for mission control."
    (and (or (org-agenda-skip-entry-if 'todo
                             '("STICKY" "NEXT" "WIP" "SOON"))
             (and (org-is-habit-p)
                  (save-excursion (or (outline-next-heading) (point-max)))))))

  (defun ec/org-agenda-skip-for-soon ()
    (let* ((is-soon (org-agenda-skip-entry-if 'todo '("SOON")))
           (skip-to-pos (or is-soon (org-agenda-skip-entry-if 'nottodo '("SOON")))))
      (and (not is-soon)
